import os,json,time,fcntl,subprocess
from pathlib import Path
p=Path('/data/matengyu/geyugong/mindhier-reproduction')
w=p/'runs/recovery-source-06e95b8'
r=p/'runs/s01-stage2-paper-seed42'
out=p/'runs/stage2-formal-supervisor'
lock=(out/'lock').open('a')
fcntl.flock(lock,fcntl.LOCK_EX|fcntl.LOCK_NB)
for entry in Path('/proc').iterdir():
 if entry.name.isdigit():
  try: cmd=(entry/'cmdline').read_bytes().replace(b'\0',b' ')
  except (FileNotFoundError,PermissionError,ProcessLookupError): continue
  if b'scripts/train_stage2_formal.py' in cmd:raise RuntimeError('Existing trainer: '+entry.name)
(out/'supervisor.pid').write_text(str(os.getpid()))
for name in ['metrics.jsonl','samples.jsonl']:
 q=r/name
 lines=q.read_text().splitlines()
 keep=[x for x in lines if json.loads(x)['step']<=3000]
 assert len(keep)==3000
 tmp=r/(name+'.resume-tmp');tmp.write_text('\n'.join(keep)+'\n');tmp.replace(q)
state={'stage':'stage2_formal_resume','formal_training':True,'supervisor_pid':os.getpid(),'resume_checkpoint_step':3000,'source_commit':'06e95b83a14b07fef51ebdb4c299b6df9914bb34','updated_at':time.time()}
(p/'runs/pipeline/state.json').write_text(json.dumps(state,indent=2))
with (r/'stdout-formal.log').open('ab') as log:
 log.write(b'\nResume after host reboot, checkpoint 3000, original source 06e95b8\n');log.flush()
 proc=subprocess.Popen([str(w/'.envs/mindhier/bin/python'),'-u','scripts/train_stage2_formal.py','--run','s01-stage2-paper-seed42','--steps','9000','--microbatch','4','--resume'],cwd=w,stdout=log,stderr=subprocess.STDOUT)
 (r/'child.pid').write_text(str(proc.pid));code=proc.wait()
with (out/'exit-codes.jsonl').open('a') as f:f.write(json.dumps({'phase':'formal_resume_3000','exit_code':code,'time':time.time()})+'\n')
state.update(stage='stage2_formal_completed' if code==0 else 'stage2_formal_resume_failed',exit_code=code,updated_at=time.time())
(p/'runs/pipeline/state.json').write_text(json.dumps(state,indent=2))
